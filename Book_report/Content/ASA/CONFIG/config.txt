## Configure interfaces

### Outside

ORACLEASA(config-if)# security-level 0
ORACLEASA(config-if)# ip address 17.17.17.1 255.255.255.252


### Inside
ORACLEASA(config)#interface GigabitEthernet1/2
ORACLEASA(config-if)# nameif inside
ORACLEASA(config-if)# security-level 100
ORACLEASA(config-if)# ip address 192.168.20.1 255.255.255.0


### DMZ
ORACLEASA(config-if)#interface GigabitEthernet1/3
ORACLEASA(config-if)# nameif dmz
ORACLEASA(config-if)# security-level 50
ORACLEASA(config-if)# ip address 192.168.30.1 255.255.255.0


## ASA DHCP for inside

ORACLEASA(config)#dhcpd address 192.168.20.100-192.168.20.150 inside
ORACLEASA(config)#dhcpd enable inside

ORACLEASA(config-network-object)# nat (inside,dmz) static 192.168.20.0
ORACLEASA(config-network-object)#
